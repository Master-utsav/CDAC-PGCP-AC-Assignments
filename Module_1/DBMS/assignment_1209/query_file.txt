delimiter //

create procedure display_hiredate_count(in y date, out x int)
begin   
       	select hiredate,

	where year(hiredate)=year(y)
	limit 1;

2. Write a function to generate an email. Call the function in select clause. email is generated as follows
​​​​​first 2 letters of name + last 2 letters of job @knowit.com

delimiter //
drop function if exists generate_email;
create function generate_email( x int) 
returns varchar(15)
begin
	declare email varchar(15);
	select concat(
		substr(ename, 1, 2), 
		substr(job, -2, 2), 
		'@knowit.com') 
	into email
	from emp
	where emp.empno = x;
        return email;
end //
delimiter ;

select empno, ename, generate_email(empno) from emp;

delimiter //
drop procedure if exists display_factorial;
create procedure display_factorial(in x int) 
begin   
	if x > 0
	then select calculate_factorial(x);
	else 
	select "Error : negative number factorial";
	end if;
end //
delimiter ;


delimiter //
drop function if exists calculate_factorial;
create function calculate_factorial(x int)
returns int
begin
	declare it int default x;
	declare fact int default 1;
	while it > 0
	do
	  set fact = fact * it;
	  set it = it - 1;
	end while;
	return fact;
end //
delimiter ;

call display_factorial(5);

4. Pass empno as a parameter to the procedure and display 'Award' as a remark if experience of employee is more than 44 years else display null for that employee.
Write a function which will calculate and return experience (no of years of services) to the procedure.

delimiter //
drop procedure if exists display_exp;
create procedure display_exp(inout x int) 
begin   
	declare experience int default 0;
	set experience = get_exp(x);
	if experience > 44 
	then select "AWARD";
	else 
	select NULL;
	end if;
end //
delimiter ;


delimiter //
drop function if exists get_exp;
create function get_exp(x int)
returns int
begin
	declare exp int default 0;
	select floor((datediff(curdate(), hiredate))/365) into exp 
	from emp
	where empno = x;
	return exp;
end //
delimiter ;

set @empno = 7369;
call display_exp(@empno); // award

set @empno = 1111;
call display_exp(@empno); // NULL
