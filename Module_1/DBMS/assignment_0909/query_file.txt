1. Find how much salary amount is invested in the employees working as a salesman of the company.

select job, sum(sal)
from emp
where job='salesman';

========================================

2. Find minimum and maximum salary earned by any employee working as a clerk in the company.

select job, min(sal), max(sal)
from emp
where job='clerk';

========================================

3. Find how many employees were hired in the year 1980.

select hiredate, count(*)
from emp
where year(hiredate)='1980'
group by hiredate;

========================================

4. Display sum of salary and minimum salary for all employees who are managers.

select sum(sal), min(sal)
from emp
where job='manager';

========================================

5. Find total salary for each dept where there is more than 1 employee working as a clerk .  Use emp table.  

select deptno, count(*)
from emp
where job='clerk'
group by deptno
having count(*)>1;

create table author_table
(
aid int auto_increment,
aname varchar(50) not null,
constraint pk_aid primary key(aid)
);

create table lang_table
(
lid int auto_increment,
lname varchar(50) not null,
constraint pk_lid primary key(lid)
);

create table book_table
(
bid int auto_increment,
bname varchar(50) not null,
lid int,
aid int,
constraint pk_bid primary key(bid),
constraint fk_lid foreign key(lid) references lang_table(lid),
constraint fk_aid foreign key(aid) references author_table(aid)
);
