2. Write a query to find the average salary for each department and then list the names of employees who earn more than the avg sal of their respective departments.

with avg_sal as (
   select deptno, avg(sal) avg_s
   from emp 
   group by deptno
)
select * 
from emp e, avg_sal av 
where e.sal > av.avg_s and e.deptno = av.deptno;

with avg_sal_dept as (
   select deptno, avg(sal) avg_s
   from emp
   group by deptno
)
select d.deptno, d.dname, av.avg_s
from dept d, avg_sal_dept av
where d.deptno = av.deptno
order by av.avg_s desc
limit 2;

Write a cursor to increment salary of all employees by 5% if employee has joined in year 1981.

delimiter // 
drop procedure if exists update_inc_sal;
create procedure update_inc_sal()
begin
     declare s decimal(7,2);
     declare eno int;
     declare y date;
     declare f int default 0;
     declare cur cursor 
	for (select empno, sal, hiredate
	     from emp_copy);
     declare continue handler for not found
	set f = 1;

     open cur;

     it:loop
	  if f = 1
	      then leave it;
	  end if;
	  fetch cur into eno, s, y;
    
          if year(y)='1981'
          then 
	       update emp_copy 
               set sal = s+(s*0.05) 
	       where empno = eno;
	  end if;
     end loop;

     close cur;

end //
delimiter ; 

call update_inc_sal();

select * from emp_copy;

3. Write a cursor to increment salary as follows
     Clerk 10 % rise
     Manager 5% rise
     Analyst 7% rise.
     No rise given for other jobs.

delimiter //
drop procedure if exists update_inc_sal;
create procedure update_inc_sal()
begin
     declare eno int;
     declare s decimal(7,2);
     declare j varchar(15);
     declare f int default 0;
     declare cur cursor 
 	for select empno, sal, job from emp_copy;
     declare continue handler for not found 
	set f = 1;
  
     open cur;
     it:loop
        fetch cur into eno, s, j;
	if f = 1
	   then leave it;
        elseif j='clerk'
	then update emp_copy
	     set sal = s + (s * 0.1)
	     where empno=eno;
	elseif j='manager'
	then update emp_copy
	     set sal = s + (s * 0.05)
	     where empno=eno;
	elseif j='analyst'
	then update emp_copy
	     set sal = s + (s * 0.07)
	     where empno=eno;
        end if;
     end loop;
     close cur;
     select * from emp_copy;
end //

delimiter ;

call update_inc_sal();

Consider following tables
product(pid, pname, price, qty) 
product_details(pid,amt)
amt = product.price * qty
Write a trigger which will calculate amt every time any update is done in the price of an item.  Hint : after update trigger

create table product_details(pid int auto_increment, amt decimal(7,2), constraint pk_pid primary key(pid));

// first trigger for inserting the data in products details

delimiter //
drop trigger if exists insert_product_price_tr;
create trigger insert_product_price_tr
before insert
on products
for each row
begin
    insert into product_details (amt)
    values(new.price * new.qty);
end //
delimiter ;

insert into products 
values(121,'chips', 45, 12);

// trigger for update the data in products details

create table product_upd_logs(pid int auto_increment, old_price decimal(7,2), new_price decimal(7,2), change_date date, constraint pk_pid primary key(pid));

delimiter //
drop trigger if exists update_product_price_tr;
create trigger update_product_price_tr
after update
on products
for each row
begin
    if new.price != old.price
    then insert into product_upd_logs(old_price, new_price, change_date)
		values(old.price, new.price, curdate());
    end if;
       
end //
delimiter ;

update products
set price = 50
where pid = 121;

select * from product_upd_logs;

after insert
on order_master
